    Before:
      fastify.get('/api/latest-read', async (request, reply) => {
        // ❌ No authentication
        // ❌ No company_id check
        return reply.send({
          success: true,
          uid: global.lastRead.uid  // ❌ Could leak other company's tags
        })

    After:
      fastify.get(
        '/api/latest-read',
        { preHandler: authenticate },  // ✅ Auth middleware
        async (request, reply) => {
          const company_id = request.user.company_id

          // ✅ Verify tag belongs to authenticated company
          const { data: lastTag } = await supabase
            .from('tag')
            .select('company_id')
            .eq('uid', global.lastRead.uid)
            .maybeSingle()

          if (!lastTag || lastTag.company_id !== company_id) {
            return reply.send({ success: false })  // ✅ No leak
          }

          return reply.send({
            success: true,
            uid: global.lastRead.uid
          })
